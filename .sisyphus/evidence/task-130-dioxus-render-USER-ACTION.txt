Recipe (PowerShell on Windows; MSVC env mandatory for any cargo step):

  # 1. MSVC env (BuildTools 14.44.35207 is the only complete toolchain)
  $msvc = "C:\BuildTools\VC\Tools\MSVC\14.44.35207"
  $env:LIB = "$msvc\lib\onecore\x64;$msvc\lib\x64;C:\Program Files (x86)\Windows Kits\10\Lib\10.0.26100.0\um\x64;C:\Program Files (x86)\Windows Kits\10\Lib\10.0.26100.0\ucrt\x64"
  $env:INCLUDE = "$msvc\include;C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\shared;C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\um;C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\ucrt;C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\winrt"
  $env:PATH = "$msvc\bin\Hostx64\x64;$env:PATH"
  $env:CC = "$msvc\bin\Hostx64\x64\cl.exe"; $env:CXX = $env:CC

  # 2. Build the counter example for wasm32 (release)
  cd C:\Users\vsbb1\source\repos\buff
  cargo build -p buff-ui-dioxus --example counter --release --target wasm32-unknown-unknown

  # 3. Post-process with wasm-bindgen 0.2.126 (must match wasm-bindgen.exe)
  $dist = "$env:TEMP\opencode\t130-render"
  Remove-Item -Recurse -Force $dist -ErrorAction SilentlyContinue
  New-Item -ItemType Directory -Path $dist | Out-Null
  wasm-bindgen --target web --out-dir $dist `
    target\wasm32-unknown-unknown\release\examples\counter.wasm

  # 4. Minimal harness (index.html)
  @"
  <!doctype html><html><head><meta charset="utf-8"><title>T130</title></head>
  <body><div id="main"></div>
  <script type="module">
    import init from './counter.js';
    await init({ module_or_path: './counter_bg.wasm' });
  </script></body></html>
  "@ | Set-Content -Encoding UTF8 $dist\index.html

  # 5. Serve (background) and open
  $ErrorActionPreference='SilentlyContinue'
  Stop-Process -Name python -ErrorAction SilentlyContinue
  Push-Location $dist
  Start-Process python -ArgumentList '-m','http.server','8765' -NoNewWindow
  Start-Sleep -Seconds 1
  Start-Process "http://localhost:8765/index.html"
  Pop-Location

  # 6. Expected: button reads "Increment (count: 0)"; click -> "(count: 1)".
  #    Screenshot to .sisyphus/evidence/task-130-dioxus-render.png on success.
